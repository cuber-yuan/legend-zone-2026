import json
import random

def main():
    data = json.loads(input())
    my_side = data["requests"][0]["mySide"]
    previous = data["responses"][-1] if data["responses"] else [-1, -1]
    actions = []
    for last_action in previous:
        # A tank cannot fire in consecutive rounds.
        if last_action >= 4:
            actions.append(-1)
        else:
            actions.append(random.choice([-1, 4, 5, 6, 7]))
    print(json.dumps({"response": actions}))

if __name__ == "__main__":
    main()
